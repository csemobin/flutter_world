widgets/image_handler.dart
import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';

class ImageHandler {
  static final _imagePicker = ImagePicker();

  // Camera Image Picker
  static Future cameraImagePicker() async {
    return await _imagePicker.pickImage(
      source: ImageSource.camera,
      imageQuality: 80,
      maxHeight: 800,
      maxWidth: 800,
    );
  }

  // Gallery Image Picker
  static Future galleryImagePicker() async {
    return await _imagePicker.pickImage(
      source: ImageSource.gallery,
      maxWidth: 800,
      maxHeight: 800,
      imageQuality: 80,
      preferredCameraDevice: CameraDevice.front,
      requestFullMetadata: true,
    );
  }

  // Bottom Sheet Image Selector
  static void bottomImageSelector(
    BuildContext context, {
    Color? backgroundColor,
    double? height,
    double? borderRadius,
    double? spacing,
    EdgeInsets? padding,
    MainAxisAlignment? alignment,
    // Camera options
    bool showCamera = true,
    String? cameraLabel,
    IconData? cameraIcon,
    String? cameraImagePath,
    double? cameraIconSize,
    Color? cameraIconColor,
    TextStyle? cameraTextStyle,
    // Gallery options
    bool showGallery = true,
    String? galleryLabel,
    IconData? galleryIcon,
    String? galleryImagePath,
    double? galleryIconSize,
    Color? galleryIconColor,
    TextStyle? galleryTextStyle,
    // General styling
    double? iconSize,
    Color? iconColor,
    TextStyle? textStyle,
    double? spaceBetweenIconAndText,
    required Function(XFile file) onImageSelected,
  }) {
    showBottomSheet(
      backgroundColor: backgroundColor ?? Colors.white,
      context: context,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(
          top: Radius.circular(borderRadius ?? 20),
        ),
      ),
      builder: (context) {
        return Container(
          height: height ?? 160,
          padding: padding ?? const EdgeInsets.symmetric(vertical: 20),
          child: Row(
            mainAxisAlignment: alignment ?? MainAxisAlignment.spaceEvenly,
            children: [
              // Camera option
              if (showCamera)
                GestureDetector(
                  onTap: () async {
                    final image = await cameraImagePicker();
                    if (image != null) {
                      onImageSelected(image);
                    }
                    Navigator.pop(context);
                  },
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      if (cameraImagePath != null)
                        Image.asset(
                          cameraImagePath,
                          width: cameraIconSize ?? iconSize ?? 40,
                          height: cameraIconSize ?? iconSize ?? 40,
                          fit: BoxFit.cover,
                        )
                      else
                        Icon(
                          cameraIcon ?? Icons.camera_alt,
                          size: cameraIconSize ?? iconSize ?? 40,
                          color: cameraIconColor ?? iconColor,
                        ),
                      SizedBox(height: spaceBetweenIconAndText ?? 8),
                      Text(
                        cameraLabel ?? "Camera",
                        style: cameraTextStyle ?? textStyle,
                      ),
                    ],
                  ),
                ),
              if (showCamera && showGallery)
                SizedBox(width: spacing ?? 0),
              // Gallery option
              if (showGallery)
                GestureDetector(
                  onTap: () async {
                    final image = await galleryImagePicker();
                    if (image != null) {
                      onImageSelected(image);
                    }
                    Navigator.pop(context);
                  },
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      if (galleryImagePath != null)
                        Image.asset(
                          galleryImagePath,
                          width: galleryIconSize ?? iconSize ?? 40,
                          height: galleryIconSize ?? iconSize ?? 40,
                          fit: BoxFit.cover,
                        )
                      else
                        Icon(
                          galleryIcon ?? Icons.photo_library,
                          size: galleryIconSize ?? iconSize ?? 40,
                          color: galleryIconColor ?? iconColor,
                        ),
                      SizedBox(height: spaceBetweenIconAndText ?? 8),
                      Text(
                        galleryLabel ?? "Gallery",
                        style: galleryTextStyle ?? textStyle,
                      ),
                    ],
                  ),
                ),
            ],
          ),
        );
      },
    );
  }
}
controllers/driver_profile_controller.dart
import 'package:get/get.dart';
import 'package:image_picker/image_picker.dart';

class DriverProfileController extends GetxController {
  // Image state
  XFile? selectedImage;

  // Update image and refresh UI
  void updateSelectedImage(XFile file) {
    selectedImage = file;
    update(); // Triggers GetBuilder to rebuild
  }

  @override
  void onClose() {
    // Clean up resources
    selectedImage = null;
    super.onClose();
  }
}
screens/driver_profile_create.dart
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:flutter_screenutil/flutter_screenutil.dart';

// Usage in your widget
Stack(
  children: [
    // Image Display
    GetBuilder(
      builder: (cnt) {
        return cnt.selectedImage != null
            ? ClipRRect(
                borderRadius: BorderRadius.circular(100),
                child: Image.file(
                  File(cnt.selectedImage!.path),
                  height: 122,
                  width: 122,
                  fit: BoxFit.cover,
                ),
              )
            : Image.asset(
                Assets.images.vector.path,
                height: 122,
                width: 122,
              );
      },
    ),
    
    // Edit Button
    Positioned(
      bottom: 10.h,
      right: 15.h,
      child: GetBuilder(
        builder: (cnt) {
          return GestureDetector(
            onTap: () {
              ImageHandler.bottomImageSelector(
                context,
                onImageSelected: (file) {
                  cnt.selectedImage = file;
                  cnt.update();
                },
              );
            },
            child: Container(
              width: 35,
              height: 35,
              decoration: BoxDecoration(
                color: Color(0XffEDEDED),
                borderRadius: BorderRadius.circular(50),
              ),
              child: Image.asset(Assets.images.editPencil.path),
            ),
          );
        },
      ),
    ),
  ],
),

💡 Best Practices

✓ Always call update()

Remember to call cnt.update() after changing selectedImage to trigger UI refresh

✓ Import dart:io

Required for File class when displaying images from XFile

✓ Use ScreenUtil

Responsive sizing with flutter_screenutil (.h, .w extensions)

✓ Handle Null Safety

Always check if selectedImage != null before accessing

🔧 Troubleshooting

Image not updating after selection?

Make sure you're calling cnt.update() after setting the selectedImage value.

Bottom sheet not appearing?

Ensure you're passing the correct BuildContext and that the widget tree has a Scaffold ancestor.

Camera/Gallery not working on device?

Check that permissions are properly configured in AndroidManifest.xml and Info.plist files.

Image quality too low/high?

Adjust the imageQuality parameter (0-100) and maxWidth/maxHeight settings in the picker methods.
